import { MetadataRoute } from 'next';

export default function robots(): MetadataRoute.robots {
    return {
        rules: [
            {
                userAgent: '*',
                allow: '/',
                disallow: [
                    '/api/', 
                    '/vault/',
                    '/login'
                ],
            },
        ],
        sitemap: 'https://dannyproano.com/sitemap.xml',
    };
}
